  model = GLiREL.from_pretrained("jackboyla/glirel-large-v0")  # 
  ดาวน์โหลดโมเดลจาก HF เอง

  tokens = ["Marco", "Polo", "was", "the", "co-founder", "of", "the", "Great",
  "Khan", "."]

  relations = model.predict_relations(tokens, labels, threshold=0.0, ner=ner,
  top_k=1)
